31 >>> partition([3.1, 1.2, 5.6, 4.7], 0, 3)

34 pivot = arr[high] 

35 i = low - 1 

36 for j in range(low, high): 

37 if arr[j] >= pivot: 

38 i += 1 

39 arr[i], arr[j] = arr[j], arr[i] 

40 arr[i + 1], arr[high] = arr[high], arr[i + 1] 

41 return i + 1
